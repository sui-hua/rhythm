# Day 页面移动端迁移 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 `rhythm-mobile` 中实现 Day 执行视图，统一展示 `Task / DailyPlan / Habit`，支持日期切换、完成状态切换，以及 `Task` 的新建、编辑、删除。

**Architecture:** 先补齐移动端 Day 所需领域模型和 Supabase 服务，再用 TanStack Query 聚合三种数据源为 `DayExecutionItem`。页面层采用 Expo Router + React Native 单列执行流，Zustand 只管理 Day 页 UI 状态，服务端缓存和乐观更新全部交给 TanStack Query。

**Tech Stack:** Expo, React Native, TypeScript, Expo Router, Supabase, TanStack Query, Zustand, React Native Testing Library, Jest

export function toDateKey(date: Date) {
  const year = date.getFullYear()
  const month = String(date.getMonth() + 1).padStart(2, '0')
  const day = String(date.getDate()).padStart(2, '0')

  return `${year}-${month}-${day}`
}

export function parseClockToMinutes(clock: string | null | undefined) {
  if (!clock) return null

  const [hourText, minuteText] = clock.split(':')
  const hours = Number(hourText)
  const minutes = Number(minuteText)

  if (!Number.isFinite(hours) || !Number.isFinite(minutes)) return null

  return hours * 60 + minutes
}

export function parseIsoToDayMinutes(value: string | null | undefined) {
  if (!value) return null

  const date = new Date(value)
  if (Number.isNaN(date.getTime())) return null

  return date.getHours() * 60 + date.getMinutes()
}

export function formatMinutesAsClock(minutes: number | null) {
  if (minutes === null) return ''

  const hours = Math.floor(minutes / 60)
  const rest = minutes % 60

  return `${String(hours).padStart(2, '0')}:${String(rest).padStart(2, '0')}`
}

export function addMinutes(minutes: number | null, duration: number | null) {
  if (minutes === null || duration === null) return null
  return minutes + duration
}

```typescript
import type { TaskRecord } from '@/domain/tasks/types'
import type { DailyPlanRecord, DayExecutionItem, HabitLogRecord, HabitRecord } from './types'
import { addMinutes, formatMinutesAsClock, parseClockToMinutes, parseIsoToDayMinutes } from './time'

type BuildInput = {
  tasks: TaskRecord[]
  dailyPlans: DailyPlanRecord[]
  habits: HabitRecord[]
  habitLogs: HabitLogRecord[]
}

const sourceRank = {
  daily_plan: 0,
  task: 1,
  habit: 2
} satisfies Record<DayExecutionItem['source'], number>

export function isDailyPlanCompleted(status: DailyPlanRecord['status']) {
  return status === 1 || status === true || status === 'done'
}

export function buildDayExecutionItems(input: BuildInput) {
  const habitLogIds = new Set(input.habitLogs.map((log) => log.habitId))

  const taskItems = input.tasks.map(toTaskItem)
  const dailyPlanItems = input.dailyPlans.map(toDailyPlanItem)
  const habitItems = input.habits.map((habit) => toHabitItem(habit, habitLogIds.has(habit.id)))

  return [...taskItems, ...dailyPlanItems, ...habitItems].sort(compareExecutionItems)
}

export function splitExecutionItems(items: DayExecutionItem[]) {
  return {
    scheduledItems: items.filter((item) => item.startMinutes !== null),
    unscheduledItems: items.filter((item) => item.startMinutes === null)
  }
}

function toTaskItem(task: TaskRecord): DayExecutionItem {
  const startMinutes = parseIsoToDayMinutes(task.startAt)
  const endMinutes = parseIsoToDayMinutes(task.endAt)

  return {
    id: task.id,
    source: 'task',
    title: task.title,
    description: task.notes,
    startAt: task.startAt,
    endAt: task.endAt,
    startMinutes,
    durationMinutes: startMinutes !== null && endMinutes !== null ? endMinutes - startMinutes : null,
    completed: task.completed,
    categoryLabel: '个人任务',
    readonly: false,
    original: task
  }
}

function toDailyPlanItem(plan: DailyPlanRecord): DayExecutionItem {
  const inheritedTime = plan.taskTime ?? plan.monthlyPlan?.taskTime ?? plan.monthlyPlan?.plan?.taskTime ?? null
  const inheritedDuration = plan.duration ?? plan.monthlyPlan?.duration ?? plan.monthlyPlan?.plan?.duration ?? null
  const startMinutes = parseClockToMinutes(inheritedTime)
  const endMinutes = addMinutes(startMinutes, inheritedDuration)

  return {
    id: plan.id,
    source: 'daily_plan',
    title: plan.title,
    description: plan.description,
    startAt: startMinutes === null ? null : formatMinutesAsClock(startMinutes),
    endAt: endMinutes === null ? null : formatMinutesAsClock(endMinutes),
    startMinutes,
    durationMinutes: inheritedDuration,
    completed: isDailyPlanCompleted(plan.status),
    categoryLabel: '今日计划',
    readonly: true,
    original: plan
  }
}

function toHabitItem(habit: HabitRecord, completed: boolean): DayExecutionItem {
  const startMinutes = parseClockToMinutes(habit.taskTime)
  const duration = habit.duration ?? (startMinutes === null ? null : 30)
  const endMinutes = addMinutes(startMinutes, duration)

  return {
    id: habit.id,
    source: 'habit',
    title: habit.title,
    description: habit.targetValue ? `目标：${habit.targetValue}` : '',
    startAt: startMinutes === null ? null : formatMinutesAsClock(startMinutes),
    endAt: endMinutes === null ? null : formatMinutesAsClock(endMinutes),
    startMinutes,
    durationMinutes: duration,
    completed,
    categoryLabel: '日常习惯',
    readonly: true,
    original: habit
  }
}

function compareExecutionItems(a: DayExecutionItem, b: DayExecutionItem) {
  if (a.startMinutes !== null && b.startMinutes !== null) {
    if (a.startMinutes !== b.startMinutes) return a.startMinutes - b.startMinutes
    return sourceRank[a.source] - sourceRank[b.source]
  }

  if (a.startMinutes !== null) return -1
  if (b.startMinutes !== null) return 1

  return sourceRank[a.source] - sourceRank[b.source]
}
```

import type { TaskFormInput, TaskRecord } from '@/domain/tasks/types'
import { validateTaskForm } from '@/domain/tasks/taskForm'
import { supabase } from './client'

function mapRow(row: any): TaskRecord {
  return {
    id: row.id,
    title: row.title,
    startAt: row.start_time,
    endAt: row.end_time,
    notes: row.description ?? '',
    completed: Boolean(row.completed)
  }
}

function toPayload(input: TaskFormInput) {
  const validated = validateTaskForm(input)

  return {
    title: validated.title,
    start_time: validated.startAt,
    end_time: validated.endAt,
    description: validated.notes,
    completed: validated.completed
  }
}

export async function listTasksForDate(date: Date): Promise<TaskRecord[]> {
  const start = new Date(date)
  start.setHours(0, 0, 0, 0)

  const end = new Date(date)
  end.setHours(23, 59, 59, 999)

  const { data, error } = await supabase
    .from('tasks')
    .select('id,title,start_time,end_time,description,completed')
    .gte('start_time', start.toISOString())
    .lte('start_time', end.toISOString())
    .order('start_time', { ascending: true })

  if (error) throw error

  return (data ?? []).map(mapRow)
}

export async function createTask(input: TaskFormInput) {
  const { data, error } = await supabase
    .from('tasks')
    .insert(toPayload(input))
    .select('id,title,start_time,end_time,description,completed')
    .single()

  if (error) throw error

  return mapRow(data)
}

export async function updateTask(taskId: string, input: Partial<TaskFormInput>) {
  const payload: Record<string, unknown> = {}

  if (input.title !== undefined) payload.title = input.title.trim()
  if (input.startAt !== undefined) payload.start_time = input.startAt
  if (input.endAt !== undefined) payload.end_time = input.endAt
  if (input.notes !== undefined) payload.description = input.notes.trim()
  if (input.completed !== undefined) payload.completed = input.completed

  const { data, error } = await supabase
    .from('tasks')
    .update(payload)
    .eq('id', taskId)
    .select('id,title,start_time,end_time,description,completed')
    .single()

  if (error) throw error

  return mapRow(data)
}

export async function deleteTask(taskId: string) {
  const { error } = await supabase.from('tasks').delete().eq('id', taskId)
  if (error) throw error
}

export async function toggleTaskComplete(taskId: string, completed: boolean) {
  return updateTask(taskId, { completed })
}

import { supabase } from './client'

function mapFrequency(value: unknown): HabitFrequency {
  if (value && typeof value === 'object' && 'type' in value) {
    return value as HabitFrequency
  }

  return { type: 'daily' }
}

function mapHabit(row: any): HabitRecord {
  return {
    id: row.id,
    title: row.title,
    targetValue: row.target_value ?? '',
    taskTime: row.task_time,
    duration: row.duration,
    frequency: mapFrequency(row.frequency),
    isArchived: Boolean(row.is_archived ?? row.archived)
  }
}

function mapLog(row: any): HabitLogRecord {
  return {
    id: row.id,
    habitId: row.habit_id,
    completedAt: row.completed_at
  }
}

export async function listHabitsForDate(_date: Date) {
  const { data, error } = await supabase
    .from('habits')
    .select('id,title,target_value,task_time,duration,frequency,archived,is_archived')
    .order('created_at', { ascending: true })

  if (error) throw error

  return (data ?? []).map(mapHabit).filter((habit) => !habit.isArchived)
}

export async function listHabitLogsForDate(date: Date) {
  const start = new Date(date)
  start.setHours(0, 0, 0, 0)

  const end = new Date(date)
  end.setHours(23, 59, 59, 999)

  const { data, error } = await supabase
    .from('habit_logs')
    .select('id,habit_id,completed_at')
    .gte('completed_at', start.toISOString())
    .lte('completed_at', end.toISOString())

  if (error) throw error

  return (data ?? []).map(mapLog)
}

export async function createHabitLog(habitId: string, date: Date) {

      log: '',
      completed_at: date.toISOString()
    })
    .select('id,habit_id,completed_at')
    .single()

  if (error) throw error

  return mapLog(data)
}

import { useMutation, useQueryClient } from '@tanstack/react-query'
import type { DayExecutionItem, HabitLogRecord } from '@/domain/day/types'
import { toDateKey } from '@/domain/day/time'
import { toggleDailyPlanComplete } from '@/services/supabase/dailyPlans'
import { createHabitLog, deleteHabitLog, listHabitLogsForDate } from '@/services/supabase/habits'
import { toggleTaskComplete } from '@/services/supabase/tasks'

export function useExecutionItemToggle(date: Date) {
  const queryClient = useQueryClient()
  const dateKey = toDateKey(date)

  const mutation = useMutation({
    mutationFn: async (item: DayExecutionItem) => {
      const nextCompleted = !item.completed

      if (item.source === 'task') {
        await toggleTaskComplete(item.id, nextCompleted)
        return
      }

      if (item.source === 'daily_plan') {
        await toggleDailyPlanComplete(item.id, nextCompleted)
        return
      }

      const logs = await listHabitLogsForDate(date)
      const existingLog = logs.find((log) => log.habitId === item.id)

      if (nextCompleted) {
        await createHabitLog(item.id, date)
      } else if (existingLog) {
        await deleteHabitLog(existingLog.id)
      }
    },
    onMutate: async (item) => {
      await Promise.all([
        queryClient.cancelQueries({ queryKey: ['day-tasks', dateKey] }),
        queryClient.cancelQueries({ queryKey: ['day-daily-plans', dateKey] }),
        queryClient.cancelQueries({ queryKey: ['day-habit-logs', dateKey] })
      ])

      const previousHabitLogs = queryClient.getQueryData<HabitLogRecord[]>(['day-habit-logs', dateKey])

      return { previousHabitLogs }
    },
    onError: (_error, _item, context) => {
      if (context?.previousHabitLogs) {
        queryClient.setQueryData(['day-habit-logs', dateKey], context.previousHabitLogs)
      }
    },
    onSettled: () => {
      void queryClient.invalidateQueries({ queryKey: ['day-tasks', dateKey] })
      void queryClient.invalidateQueries({ queryKey: ['day-daily-plans', dateKey] })
      void queryClient.invalidateQueries({ queryKey: ['day-habit-logs', dateKey] })
    }
  })

  return {
    toggleItem: mutation.mutateAsync,
    isPending: mutation.isPending,
    error: mutation.error
  }
}

          <TextInput value={notes} onChangeText={setNotes} placeholder="补充说明" multiline style={[styles.input, styles.notes]} />

          {error ? <Text style={styles.error}>{error}</Text> : null}

          <View style={styles.actions}>
            {task ? (
              <Pressable disabled={isSaving} onPress={handleDelete} style={[styles.button, styles.deleteButton]}>
                <Text style={styles.deleteText}>删除任务</Text>
              </Pressable>
            ) : null}
            <Pressable disabled={isSaving} onPress={onClose} style={styles.button}>
              <Text style={styles.buttonText}>取消</Text>
            </Pressable>
            <Pressable disabled={isSaving} onPress={handleSave} style={[styles.button, styles.saveButton]}>
              <Text style={styles.saveText}>{isSaving ? '保存中' : '保存'}</Text>
            </Pressable>
          </View>
        </View>
      </View>
    </Modal>
  )
}

function toClock(value: string | null) {
  if (!value) return null
  const date = new Date(value)
  if (Number.isNaN(date.getTime())) return null
  return `${String(date.getHours()).padStart(2, '0')}:${String(date.getMinutes()).padStart(2, '0')}`
}

function mergeDateAndClock(date: Date, clock: string) {
  const [hourText, minuteText] = clock.split(':')
  const next = new Date(date)
  next.setHours(Number(hourText), Number(minuteText), 0, 0)
  return next.toISOString()
}

import type { TaskFormInput } from '@/domain/tasks/types'

export function DayScreen() {
  const currentDate = useDayScreenState((state) => state.currentDate)
  const setCurrentDate = useDayScreenState((state) => state.setCurrentDate)
  const editingTask = useDayScreenState((state) => state.editingTask)
  const readonlyItem = useDayScreenState((state) => state.readonlyItem)
